const chat = document.querySelector("#chat"); const input = document.querySelector("#input"); const send = document.querySelector("#send"); const clear = document.querySelector("#clear"); const welcome = document.querySelector("#welcome"); const quick = document.querySelector("#quick"); function addMessage(text, type) { const item = document.createElement("div"); item.className = `message ${type}`; item.textContent = text; chat.appendChild(item); chat.scrollTop = chat.scrollHeight; } function answer(text) { const q = text.toLowerCase(); if (q.includes("привет")) { return "Привет! 👋 Я Мой ИИ. Чем займёмся?"; } if (q.includes("как тебя зовут")) { return "Я Мой ИИ — твой помощник! 🤖"; } if (q.includes("видео") || q.includes("тикток")) { return "🎬 Идея для видео: начни с неожиданного вопроса и покажи смешной финал!"; } if (q.includes("факт") || q.includes("науч")) { return "🧠 Интересный факт: у осьминога три сердца!"; } if (q.includes("игр") || q.includes("html")) { return "🎮 Давай сделаем игру на HTML! Например, нажимай на появляющиеся мишени и набирай очки."; } if (q.includes("учёб") || q.includes("учеб") || q.includes("задан")) { return "📚 Напиши школьное задание, и я помогу разобраться."; } return "🤖 Я работаю без API и пока отвечаю по встроенным правилам. Попробуй спросить про видео, игру, учёбу или интересный факт."; } function sendMessage(text) { const message = (text ?? input.value).trim(); if (!message) return; if (welcome) welcome.remove(); if (quick) quick.remove(); addMessage("Ты: " + message, "user"); addMessage("Мой ИИ: " + answer(message), "assistant"); input.value = ""; input.focus(); } send.addEventListener("click", () => sendMessage()); input.addEventListener("keydown", (event) => { if (event.key === "Enter" && !event.shiftKey) { event.preventDefault(); sendMessage(); } }); clear.addEventListener("click", () => { chat.replaceChildren(); input.value = ""; }); if (quick) { quick.querySelectorAll("button[data-q]").forEach((button) => { button.addEventListener("click", () => { sendMessage(button.dataset.q); }); }); }
